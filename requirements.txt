# Runtime dependencies of the code that exists today. Grows one slice at a time.
#
# The [validation] extra pulls in jsonschema, which core/normalise.py needs to
# check every CBOM against the official CycloneDX 1.6 schema before returning
# it. Without it the normaliser refuses to emit rather than emit unchecked.
cyclonedx-python-lib[validation]>=11.12,<12
pydantic>=2.11,<3

# Certificate parsing (scanners/container) and Ed25519 pack signatures
# (policy/sign.py). Was previously satisfied only TRANSITIVELY, via semgrep ->
# pyOpenSSL -> cryptography; a direct import needs a direct requirement, or a
# clean machine reproduces by luck.
cryptography>=42,<51

# Knowledge packs and policy packs are YAML, and four modules in the scan path
# read them (scanners/container, policy/engine, policy/apply, kpi/harness).
# It was listed only in requirements-dev.txt, which would have made a
# production install fail on the first scan.
pyyaml>=6.0

# The API and its persistence.
fastapi>=0.115,<1
sqlalchemy>=2.0,<3
uvicorn>=0.30,<1

# Scanner A (source). The scanner shells out to the `semgrep` binary rather
# than importing the package -- the CLI's --json contract is stable across
# releases in a way the internal API is not (ADR-0004).
#
# PINNED, not floored (ADR-0017). "Same input + same knowledge packs ->
# byte-identical CBOM" depends on the matching engine, so a range makes the
# determinism guarantee a range too. This is the exact version the rule pack
# and the recall/precision fixtures were scored against. It must stay equal to
# scanners.source.PINNED_SEMGREP_VERSION, which a test asserts; a scan run
# against any other version records the mismatch on its row.
semgrep==1.176.1

# Scanner D (binary). Parses ELF and PE headers, sections and symbol tables.
# Pinned to a major version: LIEF's Python API has changed shape across
# releases (the 0.x -> 1.x move renamed accessors the scanner depends on), and
# an unpinned parser is an engine whose behaviour can change under a scan --
# the thing ADR-0017 pinned semgrep to avoid.
lief>=1.0,<2

# Report generation (reports/). Pure-Python wheel with no system libraries --
# which is why it, and not weasyprint, is the choice here: weasyprint needs
# cairo/pango installed at the OS level, and a report generator that cannot run
# on an air-gapped box is no use to the estates ECDAT is for.
reportlab>=4.2,<5
